import React, { useState } from "react";
import { motion } from "framer-motion";

const ContactUs = ({ theme }) => {
  const isDark = theme === "dark";
  const [result, setResult] = useState("");
  const [isSubmitting, setIsSubmitting] = useState(false);

  const onSubmit = async (event) => {
    event.preventDefault();
    setIsSubmitting(true);
    setResult("Sending message...");

    const formData = new FormData(event.target);

    // Updated with your active Web3Forms token linked to contact.primedigitalmedia@gmail.com
    formData.append("access_key", "9a7fdbf1-3c35-412c-ba63-de5d423ddaa8");

    // Email delivery optimization fields
    formData.append("subject", "New Website Contact Inquiry");
    formData.append("from_name", "Prime Digital Media Contact Form");

    try {
      const response = await fetch("https://api.web3forms.com/submit", {
        method: "POST",
        body: formData,
      });

      const data = await response.json();

      if (data.success) {
        setResult("Thank you! Your message has been sent successfully.");
        event.target.reset();
      } else {
        console.error("Web3Forms Error:", data);
        setResult(data.message || "Something went wrong. Please try again.");
      }
    } catch (error) {
      console.error("Submission error:", error);
      setResult("Network error. Please check your connection and try again.");
    } finally {
      setIsSubmitting(false);
    }
  };

    <section
      id="contact-us"
      className={`relative py-12 px-6 sm:px-12 lg:px-24 transition-colors duration-500 overflow-hidden ${

      />
      <motion.div 
        animate={{
          scale: [1, 1.3, 1],
          opacity: [0.1, 0.2, 0.1]
        }}
        transition={{
          duration: 10,
          repeat: Infinity,
          ease: "easeInOut",
          delay: 2
        }}
        className="absolute bottom-1/4 right-1/4 translate-x-1/2 translate-y-1/2 w-[500px] h-[500px] bg-blue-600/15 blur-[160px] rounded-full pointer-events-none" 
      />

      <div className="relative z-10 max-w-7xl mx-auto">
        <div className="grid lg:grid-cols-2 gap-12 items-start">
          
          {/* LEFT COLUMN: CONTACT DETAILS */}

            transition={{ duration: 0.8 }}

            className="space-y-6"
          >
            <div>
              <span className="text-cyan-400 font-semibold tracking-wider uppercase text-sm">
                Get In Touch
              </span>
              <h2 className="text-4xl font-extrabold mt-1 tracking-tight">
                Let's Start Your <span className="text-cyan-500">Growth Journey</span>
              </h2>
              <p className={`mt-3 text-sm leading-relaxed ${isDark ? "text-gray-400" : "text-gray-600"}`}>
                We help brands scale through digital marketing, political campaign strategy, performance advertising, and creative digital solutions. Reach out to our team directly.
              </p>
            </div>

            <div className={`space-y-4 text-sm ${isDark ? "text-gray-300" : "text-gray-700"}`}>

                    className="hover:text-cyan-400 transition block"
                  >
                    contact.primedigitalmedia@gmail.com

            </div>

                  }`}
                >
                  {result}
                </motion.p>
              )}
            </form>
          </motion.div>

        </div>
      </div>
    </section>
